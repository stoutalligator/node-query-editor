Node Extract

NXQL — a SQL-like query language for XML files — press Ctrl+Enter to run

Language Reference

EXTRACT Start a query block
ROOT //Tag AS alias Find all matching root elements in the document
INTO alias//Tag AS alias Descend into descendants of the previous step
WHERE @attr = ‘val’ Filter a step — =  !=  <  >  IN (…)  NOT IN (…)
alias.@attr Select a specific attribute
alias.* Select all attributes from a step
alias//Tag WHERE … RETURN @attr AS col Inline lookup — pull a side-branch value per row
FROM ‘path/to/file.xml’ Load and query a specific XML file inline
FROM DIR ‘path/to/dir’ Run against all *.xml in a directory; adds _source column
WITH name AS (EXTRACT …) Named subquery (CTE) for multi-step logic
JOIN / LEFT JOIN … ON … Join two CTEs on one or more column pairs
LIMIT n Cap total rows returned
-- comment Single-line comments are ignored

Basic 
EXTRACT
  ROOT  //ObservationNetworkBO    AS net
  INTO  net//WeatherStationBO     AS stn
  INTO  stn//ObservationPeriodBO  AS pd
  INTO  pd//MeasurementBO         AS msr
  SELECT net.@networkId, net.@region, stn.@stationId, pd.@startDate, msr.*
LIMIT 500
Inline source + lookups 
-- Specify the XML file directly in the query.
-- Use RETURN to pull side-branch property values per row.
EXTRACT FROM 'C:/data/observations-2024.xml'
  ROOT  //ObservationNetworkBO AS net
  INTO  net//WeatherStationBO AS stn
  INTO  stn//ObservationPeriodBO AS pd
  INTO  pd//MeasurementGroupBO AS mg  WHERE @groupType = 'Atmospheric'
  INTO  mg//MeasurementBO AS msr      WHERE @measurementType IN ('WindSpeed', 'Pressure')
  SELECT
    net.@networkId,
    net.@region,
    stn.@stationId,
    stn.@stationType,
    pd.@startDate,
    pd.@endDate,
    mg//PropertyBO WHERE @propertyType = 'StationClass'  RETURN @value        AS stationClass,
    mg//PropertyBO WHERE @propertyType = 'CalibrationId' RETURN @integerValue AS calibrationId,
    msr.*
LIMIT 1000
Multi-file directory scan 
-- Run the same query against every *.xml in a folder.
-- Results are unioned; a _source column shows which file each row came from.
EXTRACT FROM DIR 'C:/data/archive/2024/'
  ROOT  //ObservationNetworkBO AS net
  INTO  net//WeatherStationBO AS stn
  INTO  stn//ObservationPeriodBO AS pd
  INTO  pd//MeasurementGroupBO AS mg
  INTO  mg//MeasurementBO AS msr
  SELECT
    net.@networkId,
    net.@region,
    stn.@stationId,
    stn.@stationType,
    stn.@elevation,
    pd.@startDate,
    mg//PropertyBO WHERE @propertyType = 'StationClass' RETURN @value AS stationClass,
    msr.*
LIMIT 2000
CTE + JOIN — wind vs temperature 
-- Join two filtered extracts side by side.
WITH wind AS (
  EXTRACT
    ROOT  //ObservationNetworkBO AS net
    INTO  net//WeatherStationBO AS stn
    INTO  stn//ObservationPeriodBO AS pd
    INTO  pd//MeasurementGroupBO AS mg   WHERE @groupType = 'Atmospheric'
    INTO  mg//MeasurementBO AS msr       WHERE @measurementType = 'WindSpeed'
    SELECT net.@networkId AS networkId, stn.@stationId AS stationId,
           pd.@startDate AS period, msr.@value AS windSpeed, msr.@unit AS windUnit
),
temperature AS (
  EXTRACT
    ROOT  //ObservationNetworkBO AS net
    INTO  net//WeatherStationBO AS stn
    INTO  stn//ObservationPeriodBO AS pd
    INTO  pd//MeasurementGroupBO AS mg   WHERE @groupType = 'Atmospheric'
    INTO  mg//MeasurementBO AS msr       WHERE @measurementType = 'Temperature'
    SELECT net.@networkId AS networkId, stn.@stationId AS stationId,
           pd.@startDate AS period, msr.@value AS temperature, msr.@unit AS tempUnit
)
SELECT w.networkId, w.stationId, w.period, w.windSpeed, t.temperature
FROM wind AS w
LEFT JOIN temperature AS t ON w.networkId  = t.networkId
                          AND w.stationId  = t.stationId
                          AND w.period     = t.period
LIMIT 500
