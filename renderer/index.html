Node Extract

Open an XML file, write a query, press Ctrl+Enter to run.

Example queries

EXTRACT
  ROOT  //*/ClaimGroupBO         AS cg
  INTO  cg//ClaimPeriodBO        AS cp
  INTO  cp//RateDetailBO         AS rd
  SELECT cg.@policyNumber, cp.@startDate, rd.*
WITH fire AS (
  EXTRACT
    ROOT //*/ClaimGroupBO          AS cg
    INTO cg//ClaimPeriodBO         AS cp
    INTO cp//RateComponentVectorBO AS rcv WHERE @rateComponentType = 'FIRE'
    INTO rcv//RateDetailBO         AS rd
    SELECT cg.@policyNumber AS policyNumber, cp.@startDate AS startDate, rd.@rateFactor AS fireFactor
),
wind AS (
  EXTRACT
    ROOT //*/ClaimGroupBO          AS cg
    INTO cg//ClaimPeriodBO         AS cp
    INTO cp//RateComponentVectorBO AS rcv WHERE @rateComponentType = 'WIND'
    INTO rcv//RateDetailBO         AS rd
    SELECT cg.@policyNumber AS policyNumber, cp.@startDate AS startDate, rd.@rateFactor AS windFactor
)
SELECT f.policyNumber, f.startDate, f.fireFactor, w.windFactor
FROM fire f
LEFT JOIN wind w ON f.policyNumber = w.policyNumber
               AND f.startDate     = w.startDate
